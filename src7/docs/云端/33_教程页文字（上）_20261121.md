`--fix` 改完方括号里的行号以后，让地址跟上：

```ts
/**
 * 让 [文字 · 文件:行号](文件#L行号) 的地址跟着方括号里的行号走。
 * @param line 修过行内引用的一行。
 * @returns 地址和文字行号一致的一行。
 */
function syncDocLinks(line: string): string {
  //M 方括号里最后一个“:行号”就是这条链接讲的位置。
  const pattern = /\[([^\]\r\n]*):(\d+)\]\(([^()\s]*)#L\d+\)/gu;
  return line.replace(pattern, (_all: string, label: string, digits: string, url: string) => {
    return '[' + label + ':' + digits + '](' + url + '#L' + digits + ')';
  });
}
```

### 4.3 改法
