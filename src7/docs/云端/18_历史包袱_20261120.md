 * 取命令行上的 --名字 值；没给就用 fallback。
 * @throws --名字 后面缺值时抛错。
 */
function getOption(name: string, fallback: string): string {
  const flag = '--' + name;
  const args = process.argv.slice(2);
  const at = args.indexOf(flag);
  if (at === -1) return fallback;

  }
  return value;
}
