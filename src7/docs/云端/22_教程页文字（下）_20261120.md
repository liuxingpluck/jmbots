```ts
/**
 * 说明.md 第一屏里的改动记录：标题和第一个 ## 之间、以 x.y.z 开头的段落。
 * @param lines 说明.md 按行切好的内容。
 * @returns 每处的行号，从 1 起。
 */
function getIntroChangelog(lines: string[]): number[] {
  const found: number[] = [];

  //M 找第一个二级标题；一个都没有就整份都算第一屏。
  let end = lines.findIndex((line) => line.startsWith('## '));
  if (end < 0) {
    end = lines.length;
  }

  //M 第一屏里以版本号开头的段落都记下。
  for (let index = 1; index < end; index += 1) {
    const line = lines[index] ?? '';
    if (/^\d+\.\d+\.\d+ /u.test(line)) {
      found.push(index + 1);
    }
  }
  return found;
}
```

第二条：`src7/plugins/**/*.ts` 里匹配 `^\s*//[MGT]?\s*\d+\.\d+\.\d+[ ：:]` 的行，提示“改动记录写进说明的「补充：改版记录」”。`//退场：` 这类有用途的标记不受影响。
