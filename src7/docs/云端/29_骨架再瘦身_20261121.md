 * 属性 5 个以上、却有一行放了两个以上属性的对象：多半是为了少算行数挤的（CLAUDE.md“对象一个属性一行”）。
 * @param source 主线的一个文件。
 * @returns 每处“文件:行号”。
 */
function findSqueezed(source: ts.SourceFile): string[] {
  const found: string[] = [];
  const visit = (node: ts.Node): void => {
    ts.forEachChild(node, visit);
    if (!ts.isObjectLiteralExpression(node)) return;
    if (node.properties.length < 5) return;

    const lines = new Set<number>();
    for (const item of node.properties) {
      const start = item.getStart(source);
      const line = source.getLineAndCharacterOfPosition(start).line;
      lines.add(line);
    }
    if (lines.size === node.properties.length) return;

    const head = source.getLineAndCharacterOfPosition(node.getStart(source)).line + 1;
    found.push(`src7/${getRelative(source.fileName)}:${String(head)}`);
  };
  visit(source);
  return found;
}
