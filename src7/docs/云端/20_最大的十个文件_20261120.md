```ts
/** 数字的写法。 */
export interface NumberRule {
  /** 没写时用的值。 */
  fallback: number;
  /** 最小值（含）。 */
  min: number;
  /** 最大值（含）；不写是不限。 */
  max?: number;
  /** 要不要整数。 */
  integer: boolean;
}

/**
 * 要一个数：没写用默认值，写了要在范围里。
 * @param value 原始值。
 * @param label 是配置里的哪一项，写进错误信息。
 * @param rule 默认值、范围、要不要整数。
 */
export function checkNumber(value: unknown, label: string, rule: NumberRule): number {
  //M 没写就用默认值。
  if (value === undefined) return rule.fallback;

  //M 报错时说清要整数还是小数、范围多大。
  let shape = '数';
  if (rule.integer) {
    shape = '整数';
  }
  let range = `不小于 ${String(rule.min)} 的`;
  if (rule.max !== undefined) {
    range = `${String(rule.min)} 到 ${String(rule.max)} 的`;
  }
  const reason = `${label} 要写成${range}${shape}`;

  //G 不是数。
  if (typeof value !== 'number' || !Number.isFinite(value)) {
    throw new Error(reason);
  }
  //G 要整数却写了小数，或者出了范围。
  const notInteger = rule.integer && !Number.isSafeInteger(value);
  const tooBig = rule.max !== undefined && value > rule.max;
  if (notInteger || value < rule.min || tooBig) {
    throw new Error(reason);
  }
  return value;
}


/** 要 true 或 false；没写用 fallback。 */
export function checkFlag(value: unknown, label: string, fallback: boolean): boolean {
  if (value === undefined) return fallback;

  if (typeof value !== 'boolean') {
    throw new Error(`${label} 要写成 true 或 false`);
  }
  return value;
}


/** 要“供应商名/模型 ID”，供应商要在 providers 里。 */
export function checkModelRef(value: unknown, label: string, providers: Record<string, unknown>): string {
  const text = checkNonEmptyText(value, label);
  const slash = text.indexOf('/');

  //G 没有斜杠，或者斜杠前的供应商不在 providers 里。
  if (slash <= 0 || providers[text.slice(0, slash)] === undefined) {
    throw new Error(`${label} 要写成“供应商名/模型 ID”，供应商要在 providers 里`);
  }
  return text;
}
```
